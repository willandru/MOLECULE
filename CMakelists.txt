cmake_minimum_required(VERSION 3.20)

project(MOLECULE
    VERSION 1.0
    LANGUAGES C CXX
)


# ============================================================
# C++ STANDARD
# ============================================================

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)


# ============================================================
# GLFW
# ============================================================

set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

add_subdirectory(
    externals/glfw-3.4
)


# ============================================================
# GLAD
# ============================================================

add_library(glad STATIC

    externals/glad/src/glad.c
)

target_include_directories(glad PUBLIC

    externals/glad/include
)


# ============================================================
# GLM
# ============================================================

add_library(glm INTERFACE)

target_include_directories(glm INTERFACE

    externals/glm
)


# ============================================================
# SOURCE FILES
# ============================================================

file(GLOB_RECURSE MOLECULE_SOURCES

    CONFIGURE_DEPENDS

    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)


# ============================================================
# MOLECULE
# ============================================================

add_executable(MOLECULE

    ${MOLECULE_SOURCES}
)


# ============================================================
# INCLUDE DIRECTORIES
# ============================================================

file(GLOB_RECURSE MOLECULE_INCLUDE_DIRS

    LIST_DIRECTORIES true

    "${CMAKE_CURRENT_SOURCE_DIR}/src/*"
)

target_include_directories(MOLECULE PRIVATE

    src

    ${MOLECULE_INCLUDE_DIRS}

    externals/glad/include

    externals/glm
)


# ============================================================
# SHADERS
# ============================================================

file(GLOB_RECURSE MOLECULE_SHADERS

    CONFIGURE_DEPENDS

    "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/shaders/*"
)


add_custom_target(MOLECULE_SHADERS
    ALL
    DEPENDS ${MOLECULE_SHADERS}
)


# ============================================================
# COPY SHADERS TO BUILD DIRECTORY
# ============================================================

add_custom_command(
    TARGET MOLECULE
    POST_BUILD

    COMMAND ${CMAKE_COMMAND} -E make_directory
        "$<TARGET_FILE_DIR:MOLECULE>/../src/graphics/shaders"

    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/shaders"
        "$<TARGET_FILE_DIR:MOLECULE>/../src/graphics/shaders"
)


# ============================================================
# LIBRARIES
# ============================================================

target_link_libraries(MOLECULE PRIVATE

    glad
    glfw
    glm
)
